== dl_reader2
"""Stream one HTTP(S) object to stdout through pv, surviving IO-governor pauses.

It stops itself before opening the URL, so no payload moves before admission, and its command identity never
changes (no exec, no children). Every request is a Range request pinned to the object's ETag. While the governor
holds pv at its floor, the server drops the stalled connection (observed after ~40 s). The reader then keeps a
held-back reserve (1 MiB) and feeds it to pv in 16 KiB pieces; it reconnects only once a piece drains within
0.25 s, i.e. after the governor resumes pv. Reconnects are capped; past the cap, or on any changed object, it
exits nonzero and the launcher marks the stream failed.
usage: dl_reader2.py URL OFFSET TOTAL_BYTES ETAG   (ETAG 'LM:<Last-Modified>' pins servers that send no ETag)
"""
import fcntl, http.client, os, signal, sys, termios, time, urllib.error, urllib.request

url, offset, total, etag = sys.argv[1], int(sys.argv[2]), int(sys.argv[3]), sys.argv[4]
OUT, RESERVE, PIECE, MAX_RECONNECTS = 1, 1 << 20, 16384, 40
os.kill(os.getpid(), signal.SIGSTOP)


def extra_headers():
    """Optional 0600 header file ("Name: value" lines), re-read before every request so a refreshed bearer token
    takes effect on reconnect; never in argv."""
    if len(sys.argv) <= 5:
        return {}
    return dict(line.split(': ', 1) for line in open(sys.argv[5]).read().splitlines() if ': ' in line)


def log(message):
    sys.stderr.write('%s reader: %s\n' % (time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime()), message))
    sys.stderr.flush()


def unread():
    return int.from_bytes(fcntl.ioctl(OUT, termios.FIONREAD, b'\0\0\0\0'), sys.byteorder)


def write_all(data):
    view = memoryview(data)
    while view:
        view = view[os.write(OUT, view):]
