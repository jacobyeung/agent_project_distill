def admit_ready(part,cfg):
 if (OUT/f'HALTED_{part}.json').exists():raise RuntimeError('HALTED: operator review required; no automatic retry')
 from budget import admit
 try:return admit(part,cfg)
 except (AssertionError,ValueError,FileNotFoundError,FileExistsError,KeyError,StopIteration) as exc:
  write_json(OUT/f'NOT_ADMITTED_{part}.json',dict(status='NOT_ADMITTED',part=part,reason=str(exc) or type(exc).__name__,skipped=True))
  return None

  if event=='open' and isinstance(args[0],(str,bytes)):
