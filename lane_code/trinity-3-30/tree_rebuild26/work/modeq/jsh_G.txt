   231	def digest(path):
   232	    h = hashlib.sha256()
   233	    with open(path, 'rb') as stream:
   234	        for block in iter(lambda: stream.read(8 << 20), b''):
   235	            h.update(block)
   236	    return h.hexdigest()
   237	def host(p):

   242	if digest(dest/'artifact.json') != sha:
   243	    raise SystemExit('init artifact differs from its pin')
   244	artifact = json.loads((dest/'artifact.json').read_text())
   245	heads, manifest = host(artifact['heads']['path']), host(artifact['full_weights']['path'])
   246	os.link(heads, dest/'heads.pt')
   247	if digest(dest/'heads.pt') != artifact['heads']['sha256']:
   248	    raise SystemExit('init heads differ')
   249	os.link(manifest, dest/'full_weights'/manifest.name)
   250	if digest(dest/'full_weights'/manifest.name) != artifact['full_weights']['sha256']:
   251	    raise SystemExit('init manifest differs')
   252	spec = json.loads(manifest.read_text())
   253	pins = {spec['index']['name']: spec['index']['sha256'], **{f['name']: f['sha256'] for f in spec['files']}}
   254	for name, expected in pins.items():
   255	    os.link(manifest.parent/name, dest/'full_weights'/name)
   256	    if digest(dest/'full_weights'/name) != expected:
   257	        raise SystemExit('init weights differ: ' + name)
   258	PYINIT
   259	    touch "$dest.part/.verified"
   260	    mv "$dest.part" "$dest"
   261	    tk_step "init mirror verified: ${sha:0:16}"
   262	}
